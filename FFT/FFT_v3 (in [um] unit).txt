S1 Matrixing
tic
field(isnan(field)) = eps;

dimE = sqrt(length(field))+expand*2;   %每个模块的x,y像素点数量

%实空间的坐标序列

Ly = 2*abs(field(2,1))/unit;                 %实空间y方向的长度,这里要求对称取点

%k空间的坐标序列

Ez=padarray(Ez,[expand expand]);

FFT
tic
normEx=conj(Ex).*Ex;
normEy=conj(Ey).*Ey;
normEz=conj(Ez).*Ez;
normE0 = normEx + normEy + normEz;      %计算得到的实空间pattern
clear Exq Eyq Ezq
% 做场的积分归一化，使得总场强²加和为1，便于对比
fEx = fftshift(fft2(Ex));
fEy = fftshift(fft2(Ey));
fEz = fftshift(fft2(Ez));                  %对各个分量进行FFT
normfEx=fEx.*conj(fEx);
normfEy=fEy.*conj(fEy);
normfEz=fEz.*conj(fEz);
normfE0 = normfEx + normfEy + normfEz;  %k空间pattern
toc

iFFT(+滤波)
tic
K0=k0*NA;
[KX,KY]=meshgrid(kx,ky);
KZ=sqrt(k0.^2-KX.^2-KY.^2).*((KX.^2+KY.^2)<=K0^2);
phase=exp(1i*KZ*h0);
% phase=1;
ffEx=fEx.*phase.*((KX.^2+KY.^2)<=K0^2);
ffEy=fEy.*phase.*((KX.^2+KY.^2)<=K0^2);
ffEz=fEz.*phase.*((KX.^2+KY.^2)<=K0^2);
normffEx0=ffEx.*conj(ffEx);
normffEy0=ffEy.*conj(ffEy);
normffEz0=ffEz.*conj(ffEz);
normffE0 = normffEx0 + normffEy0 + normffEz0;  %k空间pattern
ifEx=ifft2(ifftshift(ffEx));
ifEy=ifft2(ifftshift(ffEy));
ifEz=ifft2(ifftshift(ffEz));
normifEx0=ifEx.*conj(ifEx);
normifEy0=ifEy.*conj(ifEy);
normifEz0=ifEz.*conj(ifEz);
normifE0 = normifEx0 + normifEy0 + normifEz0;  %k空间pattern
toc
